
# Water Jug Problem using DFS

CAP_A = 9
CAP_B = 8
GOAL = 7


# Generate all possible neighboring states
def get_neighbors(state):

    a, b = state
    neighbors = []

    # Fill either jug
    neighbors.append((CAP_A, b))
    neighbors.append((a, CAP_B))

    # Empty either jug
    neighbors.append((0, b))
    neighbors.append((a, 0))

    # Pour water from Jug A to Jug B
    pour = min(a, CAP_B - b)
    neighbors.append((a - pour, b + pour))

    # Pour water from Jug B to Jug A
    pour = min(b, CAP_A - a)
    neighbors.append((a + pour, b - pour))

    return list(dict.fromkeys(n for n in neighbors if n != state))


# Depth First Search Algorithm
def dfs(start):

    visited = {start}
    stack = [(start, [start])]

    while stack:

        state, path = stack.pop()

        # Check whether goal is reached
        if state[0] == GOAL or state[1] == GOAL:
            return path

        # Explore neighboring states
        for neighbor in get_neighbors(state):

            if neighbor not in visited:

                visited.add(neighbor)

                stack.append((neighbor, path + [neighbor]))

    return None


# Main program
if __name__ == "__main__":

    start = (0, 0)

    solution = dfs(start)

    if solution:

        print("Solution Found:\n")

        for step, state in enumerate(solution):
            print("Step", step, "-> Jug A:", state[0], ", Jug B:", state[1])

    else:
        print("No solution found.")

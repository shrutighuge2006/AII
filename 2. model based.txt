from collections import deque

# Model-Based Intelligent Agent for Grid Navigation

# Create the grid
# 0 = Free cell
# X = Obstacle

grid = [
    ['S', 0, 0, 'X'],
    [0, 'X', 0, 0],
    [0, 0, 0, 'X'],
    ['X', 0, 0, 'G']
]

# Number of rows and columns
rows = len(grid)
cols = len(grid[0])

# Start and goal positions
start = (0, 0)
goal = (3, 3)

# Queue stores current node and path
queue = deque([(start, [start])])

# Internal state (memory): visited cells
visited = set()
visited.add(start)

# Possible movements
directions = [(-1, 0), (1, 0), (0, -1), (0, 1)]

print("Grid Navigation Begins\n")

# Variable to store final path
final_path = None

# Continue until queue becomes empty
while queue:

    # Remove first element
    current, path = queue.popleft()

    # Display current state
    print("Current Position:", current)
    print("Visited:", visited)

    # Goal test
    if current == goal:
        final_path = path
        print("\nGoal Reached!")
        break

    # Explore neighbours
    for dr, dc in directions:

        # Compute neighbour coordinates
        nr = current[0] + dr
        nc = current[1] + dc

        # Check grid boundary
        if 0 <= nr < rows and 0 <= nc < cols:

            # Skip obstacles
            if grid[nr][nc] == 'X':
                continue

            # Skip visited cells
            if (nr, nc) in visited:
                continue

            # Update internal state
            visited.add((nr, nc))

            # Add neighbour to queue
            queue.append(((nr, nc), path + [(nr, nc)]))

print("\n================================")

if final_path:
    print("Path Found:")
    print(final_path)
else:
    print("No path exists.")

print("\nRational Behaviour Analysis")
print("1. The agent remembers visited cells.")
print("2. It never revisits explored locations.")
print("3. It avoids obstacles.")
print("4. It systematically searches for the goal.")
print("Hence, it behaves as a Model-Based Intelligent Agent.")
